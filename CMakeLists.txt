set(WATER_USE_OPENMP FALSE)
set(WATER_RUNTIME_DIR "")
if(NOT WATER_OPENMP STREQUAL "OFF")
  if(MSVC)
    # The preserved sources use unsigned OpenMP loops and max reductions.
    set(OpenMP_RUNTIME_MSVC llvm CACHE STRING "MSVC OpenMP runtime" FORCE)
  endif()
  find_package(OpenMP QUIET COMPONENTS CXX)
  if(OpenMP_CXX_FOUND)
    set(WATER_USE_OPENMP TRUE)
    if(MSVC)
      set(_arch x64)
      set(_suffix x86_64)
      if(CMAKE_GENERATOR_PLATFORM MATCHES "[Aa][Rr][Mm]64" OR CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64")
        set(_arch arm64)
        set(_suffix aarch64)
      elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
        set(_arch x86)
        set(_suffix i386)
      endif()
      # Discover only the selected compiler's existing Visual Studio runtime.
      # These DLLs are not copied, packaged, downloaded or redistributed.
      file(TO_CMAKE_PATH "${CMAKE_CXX_COMPILER}" _compiler)
      string(REGEX REPLACE "/VC/.*$" "" _vs_root "${_compiler}")
      file(GLOB _runtime_dirs LIST_DIRECTORIES TRUE
        "${_vs_root}/VC/Redist/MSVC/*/debug_nonredist/${_arch}/Microsoft.VC*.OpenMP.LLVM"
        "${_vs_root}/VC/Redist/MSVC/*/${_arch}/Microsoft.VC*.OpenMP.LLVM")
      list(SORT _runtime_dirs COMPARE NATURAL ORDER DESCENDING)
      find_file(WATER_OPENMP_RUNTIME_DLL NAMES "libomp140.${_suffix}.dll"
        HINTS ${_runtime_dirs} ENV PATH DOC "Existing MSVC LLVM OpenMP runtime DLL")
      if(WATER_OPENMP_RUNTIME_DLL)
        get_filename_component(WATER_RUNTIME_DIR "${WATER_OPENMP_RUNTIME_DLL}" DIRECTORY)
      else()
        set(WATER_USE_OPENMP FALSE)
        message(STATUS "MSVC LLVM OpenMP runtime DLL not found; no runtime will be copied")
      endif()
    endif()
  endif()
  if(NOT WATER_USE_OPENMP AND WATER_OPENMP STREQUAL "ON")
    message(FATAL_ERROR "OpenMP requested but compiler/runtime unavailable. Install the matching toolchain/runtime, set WATER_OPENMP_RUNTIME_DLL on Windows or OpenMP_ROOT on macOS, or explicitly use -DWATER_OPENMP=OFF for serial execution.")
  endif()
endif()
set(WATER_OPENMP_JSON false)
if(WATER_USE_OPENMP)
  set(WATER_OPENMP_JSON true)
  message(STATUS "WATER OpenMP: enabled")
else()
  message(STATUS "WATER OpenMP: disabled; existing serial source path selected")
endif()

function(water_program name source)
  if(EXISTS "${WATER_ENGINE_DIR}/${source}")
    add_executable(${name} "${WATER_ENGINE_DIR}/${source}")
    target_compile_features(${name} PRIVATE cxx_std_17)
    set_target_properties(${name} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
      target_compile_options(${name} PRIVATE /EHsc /utf-8)
      target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    if(WATER_USE_OPENMP)
      target_link_libraries(${name} PRIVATE OpenMP::OpenMP_CXX)
    endif()
  endif()
endfunction()
water_program(gallery_sim src/gallery_simulate.cpp)

water_program(trace src/trace.cpp)
# Compensated predicates require ordered floating-point operations. std::fma
# remains explicit; do not enable reassociation/fast-math for these targets.

else()
  target_compile_options(trace PRIVATE -fno-fast-math -ffp-contract=off)
endif()
